━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  מה המערכת עושה?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  המערכת רצה אוטומטית 4 פעמים בלילה (01:00, 02:00, 03:00, 04:00).
  כל ריצה נמשכת עד 5.5 דקות — סה"כ ~22 דקות הורדה אפקטיביות.
  ריצה שנייה ממשיכה בדיוק מאיפה שהראשונה עצרה (מדלגת על קבצים קיימים).

  בכל ריצה:
  1. קוראת רשימת RSS מקובץ טקסט ב-Drive.
  2. לכל פודקאסט — מורידה פרקים שפורסמו ב-7 הימים האחרונים
     ואינם קיימים עדיין בתיקייה.
  3. שומרת כל פרק בתיקייה ייעודית לפי שם הפודקאסט.
  4. יוצרת לצד כל פרק קובץ .lrc עם כל פרטי הפרק.
  5. שולחת מייל אוטומטי בשגיאה קריטית.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  קבצים במערכת
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌─ trigger_script.gs     ← מוכנס לתוך Google Apps Script (קצר בכוונה)
  ├─ podcast_processor.js  ← כל הלוגיקה — נשמר ב-Google Drive
  └─ podcasts.txt          ← רשימת RSS — יש ליצור ידנית ב-Drive


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  שלב 1 — הכנת קבצים ב-Google Drive
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  א. העלה את podcast_processor.js ל-Drive.
     (Drive → New → File upload)

  ב. צור קובץ בשם podcasts.txt עם כתובות ה-RSS:

       # שורות המתחילות ב-# הן הערות — מדולגות
       # שורות ריקות מדולגות

       https://feeds.simplecast.com/example_podcast
       https://rss.art19.com/another-show
       https://anchor.fm/s/abc123/podcast/rss

  ג. צור תיקייה שבה יישמרו הפרקים (לדוגמה: "פודקאסטים").

  ד. מצא את ה-ID של כל אחד מ-3 הפריטים:
     פתח כל קובץ/תיקייה ב-Drive — העתק מה-URL:

       קובץ:    https://drive.google.com/file/d/ [ID] /view
       תיקייה:  https://drive.google.com/drive/folders/ [ID]

     ⚠️  אין תמיכה בנתיבים (כמו /folder/file.js) ב-Google Apps Script.
         ה-ID הוא הדרך היחידה האמינה לגשת לקבצים.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  שלב 2 — הגדרת Google Apps Script
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  א. פתח: https://script.google.com → "New project"
  ב. מחק את הקוד הקיים.
  ג. הדבק את כל תוכן trigger_script.gs.
  ד. מלא את ה-IDs בראש הקובץ:

       var PROCESSOR_FILE_ID = "ID של podcast_processor.js";
       var RSS_LIST_FILE_ID  = "ID של podcasts.txt";
       var ROOT_FOLDER_ID    = "ID של תיקיית הפודקאסטים";

  ה. שמור (Ctrl+S).


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  שלב 3 — הרצה ראשונה ואישור הרשאות
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  א. בתפריט הנפתח בחר "dailyRun" ← לחץ ▶ Run.
  ב. אשר את ההרשאות הנדרשות:
       • Google Drive (קריאה וכתיבה)
       • Gmail (שליחת מייל שגיאה)
       • External requests (הורדת פרקים ו-RSS)

     אם מופיע "Google hasn't verified this app":
     לחץ "Advanced" ← "Go to [שם הפרויקט] (unsafe)"

  ג. בדוק לוג תקין (View → Logs):
       🚀 מערכת הורדת פודקאסטים — ...
       📋 נטענו N כתובות RSS.
       🎙️  ערוץ: ...
       ✅ נשמר: ...
       🏁 ריצה הסתיימה: ...


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  שלב 4 — הפעלת הטריגרים האוטומטיים
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  א. בתפריט הנפתח בחר "setupTrigger" ← לחץ ▶ Run.
  ב. פעולה זו נדרשת פעם אחת בלבד.
  ג. 4 טריגרים ייווצרו: 01:00, 02:00, 03:00, 04:00.

  לאימות: תפריט שמאל → שעון ⏰ (Triggers) — אמורים להופיע 4 שורות.

  ⚠️  לא להריץ setupTrigger שוב — ייווצרו טריגרים כפולים.
      אם בטעות הרצת שוב: כנס ל-Triggers, מחק הכל, הרץ setupTrigger מחדש.

  מבנה תוצאות ב-Drive
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  📁 פודקאסטים/
  ├── 📁 שם פודקאסט א׳/
  │    ├── 🎵 כותרת פרק 1.mp3
  │    ├── 📄 כותרת פרק 1.lrc
  │    ├── 🎵 כותרת פרק 2.mp3
  │    └── 📄 כותרת פרק 2.lrc
  └── 📁 שם פודקאסט ב׳/
       ├── 🎵 כותרת פרק.m4a
       └── 📄 כותרת פרק.lrc

  קובץ .lrc — פרטי הפרק:
  ┌────────────────────────────────────────
  │ פודקאסט:      שם הפודקאסט
  │ פרק:          כותרת הפרק המלאה
  │ תאריך פרסום:  Mon, 26 May 2025 08:00:00
  │ מגיש / כותב:  שם המגיש
  │ משך:          45:32
  │ עונה:         3  (אם קיים)
  │ מספר פרק:     12 (אם קיים)
  │
  │ ── תיאור ──────────────────────────────
  │ [תיאור מלא של הפרק, ללא תגיות HTML]
  └────────────────────────────────────────

  הורדת קבצים גדולים (מעל 45MB)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  GAS מגביל הורדה בודדת ל-50MB. הפתרון המובנה — Chunked Download:

  1. HEAD request לבדיקת גודל הקובץ.
  2. קבצים עד 45MB: הורדה רגילה.
  3. קבצים גדולים יותר (אם השרת תומך ב-Range):
     א. הורדה ב-chunks של 15MB.
     ב. כל chunk מועלה מיידית ל-Drive (Resumable Upload API).
     ג. רק 15MB בזיכרון בכל רגע — לא נשמר הקובץ כולו בבת אחת.

  הלוג יציג התקדמות:
    📦 קובץ גדול: 87MB
    📥 Chunk 1: 0MB – 15MB
    ✔ 15/87MB הועלו.
    ...

  עדכון ושינויים בעתיד
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  הוספה/הסרה של פודקאסט:
    ערוך את podcasts.txt ב-Drive — יכנס לתוקף בריצה הבאה.

  עדכון קוד הלוגיקה:
    החלף את podcast_processor.js ב-Drive — יטען אוטומטית.

  שינוי שעות הריצה:
    ערוך את הרשימה [1, 2, 3, 4] ב-setupTrigger ← הרץ שוב.

  פתרון תקלות
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ❌ "No item with the given ID could be found"
     → אחד מה-ID שגוי. בדוק שוב PROCESSOR_FILE_ID, RSS_LIST_FILE_ID, ROOT_FOLDER_ID.

  ❌ פרק לא הורד — "HTTP 403"
     → הפיד דורש מנוי/אימות — לא ניתן להוריד אוטומטית.

  ❌ "Chunk upload failed: 403"
     → בעיית הרשאות ל-Drive API. ודא שמריץ הסקריפט הוא בעל הקבצים ב-Drive.

  ❌ טריגרים כפולים
     → כנס ל-Triggers, מחק הכל ידנית, הרץ setupTrigger מחדש.

  ✅ היסטוריית ריצות ולוגים:
     Apps Script → Executions (תפריט שמאל).
